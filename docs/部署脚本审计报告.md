## 高风险项修复代码

### 修复代码 A：管理员权限校验（install.bat / install_e_wms.bat 开头添加）

```bat
REM === 管理员权限校验（AI-SEC-F01）===
net session >nul 2>&1
if errorlevel 1 (
  echo [ERROR] 请以管理员身份运行此脚本（右键 → 以管理员身份运行）。
  echo         安装 Python、写入系统路径、创建服务需要管理员权限。
  pause
  exit /b 1
)
```

### 修复代码 B：PowerShell + VC++ 运行时预检

```bat
REM === PowerShell / WMF 版本预检（AI-SEC-F01）===
for /f "delims=" %%v in ('powershell -NoProfile -Command "$PSVersionTable.PSVersion.Major" 2^>nul') do set "PS_MAJOR=%%v"
if not defined PS_MAJOR (
  echo [ERROR] PowerShell 未找到。请安装 WMF 5.1。
  echo         下载: https://www.microsoft.com/download/details.aspx?id=54616
  pause
  exit /b 1
)
if !PS_MAJOR! LSS 5 (
  echo [ERROR] PowerShell 版本 !PS_MAJOR! 过低，需要 5.0+（Expand-Archive 需要）。
  echo         请安装 WMF 5.1: https://www.microsoft.com/download/details.aspx?id=54616
  pause
  exit /b 1
)
echo [CHECK] PowerShell !PS_MAJOR! OK

REM === VC++ 运行时预检（AI-SEC-F01）===
set "VC_REDIST_KEY=HKLM\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\X64"
reg query "%VC_REDIST_KEY%" /v Version >nul 2>nul
if errorlevel 1 (
  echo [WARN] VC++ 2015-2022 x64 Redistributable 未检测到。
  echo        Python 3.11 需要此运行时。请安装: https://aka.ms/vs/17/release/vc_redist.x64.exe
  echo        继续安装可能导致 Python 启动失败。
  choice /C YN /M "是否继续"
  if errorlevel 2 exit /b 1
)
```

### 修复代码 C：文件日志记录

```bat
REM === 文件日志初始化（AI-SEC-F01）===
set "LOG_DIR=%RUN_DIR%\logs"
if not exist "%LOG_DIR%" mkdir "%LOG_DIR%"
for /f %%i in ('powershell -NoProfile -Command "Get-Date -Format yyyyMMdd_HHmmss"') do set "STAMP=%%i"
set "INSTALL_LOG=%LOG_DIR%\install_%STAMP%.log"

REM 日志函数：调用方式  call :log "消息"  或  call :logerr "错误消息"
:log
echo [%date% %time%] %~1
echo [%date% %time%] %~1>>"%INSTALL_LOG%"
goto :eof

:logerr
echo [%date% %time%] [ERROR] %~1 1>&2
echo [%date% %time%] [ERROR] %~1>>"%INSTALL_LOG%"
goto :eof
```

### 修复代码 D：回滚机制

```bat
REM === 回滚机制（AI-SEC-F01）===
set "ROLLBACK_NEEDED="

:do_rollback
if not defined ROLLBACK_NEEDED goto :eof
call :log "===== 开始回滚 ====="
REM 1. 删除已创建的服务（如存在）
if defined SERVICE_NAME (
  nssm stop "%SERVICE_NAME%" >nul 2>nul
  nssm remove "%SERVICE_NAME%" confirm >nul 2>nul
  call :log "已删除服务: %SERVICE_NAME%"
)
REM 2. 删除已复制的文件（仅非原地安装）
if not defined IN_PLACE_INSTALL (
  if exist "%INSTALL_DIR%\app.py" (
    rd /s /q "%INSTALL_DIR%" >nul 2>nul
    call :log "已清理安装目录: %INSTALL_DIR%"
  )
)
REM 3. 删除桌面快捷方式
if exist "%USERPROFILE%\Desktop\WMS.lnk" del /q "%USERPROFILE%\Desktop\WMS.lnk" >nul 2>nul
call :log "已删除桌面快捷方式"
call :log "===== 回滚完成 ====="
call :logerr "部署失败，已回滚。详见日志: %INSTALL_LOG%"
pause
exit /b 1
```

### 修复代码 E：端口预检 + 幂等性检查

```bat
REM === 端口预检（AI-SEC-F01）===
set "TARGET_PORT=8080"
powershell -NoProfile -Command "$c=Get-NetTCPConnection -LocalPort %TARGET_PORT% -State Listen -ErrorAction SilentlyContinue; if($c){Write-Host 'OCCUPIED'; exit 1} else {Write-Host 'FREE'; exit 0}"
if errorlevel 1 (
  call :logerr "端口 %TARGET_PORT% 已被占用，部署中止。请先停止占用该端口的进程。"
  pause
  exit /b 1
)
call :log "端口 %TARGET_PORT% 空闲"

REM === 幂等性检查（AI-SEC-F01）===
if exist "%RUN_DIR%\.installed.flag" (
  call :log "检测到已安装标记 %RUN_DIR%\.installed.flag，跳过重复安装。"
  call :log "如需重新安装，请先删除该标记文件并卸载旧版本。"
  pause
  exit /b 0
)
```

### 修复代码 F：Git 凭据安全（替代明文 PAT）

```bat
REM === Git 凭据安全（AI-SEC-F01）===
REM 不再硬编码 PAT，改用环境变量 GITHUB_TOKEN
if not defined GITHUB_TOKEN (
  call :logerr "环境变量 GITHUB_TOKEN 未设置。请设置后再运行部署。"
  call :log "例如: set GITHUB_TOKEN=github_pat_xxxx..."
  pause
  exit /b 1
)
REM 使用环境变量注入凭据，不写入 .git-credentials 文件
git config --global credential.helper store
REM 临时凭据通过 URL 参数注入，进程结束后不持久化
for /f "delims=" %%u in ('git remote get-url origin 2^>nul') do set "REMOTE_URL=%%u"
REM 限制 .git-credentials 文件 ACL 仅 SYSTEM 和 Administrators 可读
if exist "%USERPROFILE%\.git-credentials" (
  icacls "%USERPROFILE%\.git-credentials" /inheritance:r /grant:r "SYSTEM:F" "Administrators:F" >nul 2>nul
  call :log "已限制 .git-credentials 文件 ACL"
)
```
